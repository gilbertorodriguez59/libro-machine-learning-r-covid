3.10 Laboratorio interactivo: exploración de distribuciones

Selecciona una variable de iris y examina su distribución.

#| '!! shinylive warning !!': |

#| standalone: true
#| viewerHeight: 700

library(shiny)

ui <- fluidPage(
  titlePanel("Explorador interactivo de datos"),
  sidebarLayout(
    sidebarPanel(
      selectInput("variable", "Variable", choices = names(iris)[1:4],
                  selected = "Sepal.Length"),
      radioButtons("grafica", "Tipo de gráfica",
                   choices = c("Histograma" = "hist",
                               "Boxplot por especie" = "box",
                               "Densidad" = "dens")),
      sliderInput("clases", "Número de intervalos", 5, 30, 12)
    ),
    mainPanel(
      plotOutput("plot", height = "420px"),
      tableOutput("resumen")
    )
  )
)

server <- function(input, output, session) {
  valores <- reactive(iris[[input$variable]])

  output$plot <- renderPlot({
    x <- valores()
    if (input$grafica == "hist") {
      hist(x, breaks = input$clases,
           main = paste("Histograma de", input$variable),
           xlab = input$variable)
    } else if (input$grafica == "box") {
      boxplot(x ~ iris$Species, xlab = "Especie",
              ylab = input$variable,
              main = paste(input$variable, "por especie"))
    } else {
      plot(density(x), lwd = 3,
           main = paste("Densidad de", input$variable),
           xlab = input$variable)
      grid()
    }
  })

  output$resumen <- renderTable({
    x <- valores()
    data.frame(
      N = length(x),
      Media = round(mean(x), 3),
      Mediana = round(median(x), 3),
      Desviacion = round(sd(x), 3),
      Minimo = min(x),
      Maximo = max(x)
    )
  })
}

shinyApp(ui, server)

           main = paste("ATUS: accidentes por hora - mes", input$mes))

    } else {
      plot(d$ID_HORA, d$PORC_VICTIMAS, type = "b", pch = 19,
           xlab = "Hora del día", ylab = "% con víctimas",
           ylim = c(0, max(5, max(d$PORC_VICTIMAS, na.rm = TRUE) * 1.1)),
           main = paste("ATUS: proporción con víctimas - mes", input$mes))
    }
    grid()
  })

  output$resumen <- renderTable({
    d <- datos_mes()
    data.frame(
      Indicador = c("Accidentes del mes", "Con víctimas", "% con víctimas"),
      Valor = c(sum(d$TOTAL), sum(d$CON_VICTIMAS),
                round(100 * sum(d$CON_VICTIMAS) / sum(d$TOTAL), 2))
    )
  })

  output$interpretacion <- renderText({
    d <- datos_mes()
    h <- d$ID_HORA[which.max(d$TOTAL)]
    paste0("En el mes seleccionado, la hora con mayor número de accidentes es ",
           h, ":00. Cambia el mes y compara si el patrón se conserva.")
  })
}
